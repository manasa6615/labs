from sklearn.linear_model import LinearRegression
import matplotlib.pyplot as plt

X = [[1], [2], [3], [4], [5]]
y = [20, 35, 50, 65, 80]

model = LinearRegression()

hours = [[6]]

print("Intercept:", model.intercept_)

plt.scatter(X, y)
plt.plot(X, model.predict(X))
plt.xlabel("Hours Studied")
plt.ylabel("Exam Marks")
plt.title("Hours Studied vs Exam Marks")
plt.show()

from sklearn.cluster import KMeans
import matplotlib.pyplot as plt

points = [[1, 2], [1, 4], [2, 3],
          [8, 8], [9, 10], [8, 9]]

model = KMeans(n_clusters=2, random_state=0, n_init=10)
model.fit(points)

print("Cluster labels:", model.labels_)
print("Cluster centers:", model.cluster_centers_)

# Graph
plt.scatter(
    [p[0] for p in points],
    [p[1] for p in points],
    c=model.labels_
)

# Cluster centers
plt.scatter(
    model.cluster_centers_[:, 0],
    model.cluster_centers_[:, 1],
    marker='X',
    s=200
)

plt.xlabel("X")
plt.ylabel("Y")
plt.title("K-Means Clustering")
plt.show()

# Neural Network for XOR with Graph

import numpy as np
import matplotlib.pyplot as plt

np.random.seed(1)

X = np.array([[0, 0],
              [0, 1],
              [1, 0],
              [1, 1]])

y = np.array([[0],
              [1],
              [1],
              [0]])

W1 = np.random.randn(2, 4)
b1 = np.zeros((1, 4))

W2 = np.random.randn(4, 1)
b2 = np.zeros((1, 1))

lr = 0.5

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

# Training
for _ in range(10000):

    h = sigmoid(X @ W1 + b1)
    output = sigmoid(h @ W2 + b2)

    error = output - y

    d2 = error * output * (1 - output)

    dW2 = h.T @ d2
    db2 = np.sum(d2, axis=0, keepdims=True)

    d1 = (d2 @ W2.T) * h * (1 - h)

    dW1 = X.T @ d1
    db1 = np.sum(d1, axis=0, keepdims=True)

    W2 -= lr * dW2
    b2 -= lr * db2

    W1 -= lr * dW1
    b1 -= lr * db1

# Predictions
print("Predictions:")
print((output > 0.5).astype(int))

# Graph
plt.scatter(
    X[:, 0],
    X[:, 1],
    c=output.flatten() > 0.5,
    s=200
)

plt.xlabel("X1")
plt.ylabel("X2")
plt.title("XOR Neural Network")
plt.xticks([0, 1])
plt.yticks([0, 1])
plt.grid(True)
plt.show()

# SVM Classification with Graph

from sklearn.svm import SVC
import matplotlib.pyplot as plt
import numpy as np

X = [[1, 2], [2, 3], [3, 3],
     [6, 7], [7, 8], [8, 8]]

y = ["A", "A", "A", "B", "B", "B"]

model = SVC(kernel="linear")
model.fit(X, y)

print("Predicted class:", model.predict([[2, 2]])[0])

# Convert data to array
X = np.array(X)

# Plot points
plt.scatter(X[:3, 0], X[:3, 1], label="Class A")
plt.scatter(X[3:, 0], X[3:, 1], label="Class B")

# Plot decision boundary
w = model.coef_[0]
b = model.intercept_[0]

x_values = np.linspace(0, 9, 100)
y_values = -(w[0] * x_values + b) / w[1]

plt.plot(x_values, y_values, label="Decision Boundary")

# New prediction point
plt.scatter(2, 2, marker="X", s=150, label="New Point")

plt.xlabel("X1")
plt.ylabel("X2")
plt.title("SVM Classification")
plt.legend()
plt.grid(True)
plt.show()
